          isSelected

            ? 'border-yellow-500 bg-yellow-500/10 text-white font-semibold shadow-sm'

            : 'border-neutral-800 bg-neutral-900/60 hover:border-red-500/60 hover:bg-neutral-900 text-neutral-300'

        btn.onclick = () => selectOption(q.id, opt.value);

        btn.innerHTML = `

          <div class="w-5 h-5 rounded-full border-2 flex-shrink-0 flex items-center justify-center ${

            isSelected ? 'border-yellow-400 bg-yellow-500' : 'border-neutral-600'

          }">

            ${isSelected ? '<div class="w-2 h-2 rounded-full bg-black"></div>' : ''}

          </div>

        optionsContainer.appendChild(btn);

      document.getElementById('prev-btn').style.visibility = currentStep === 0 ? 'hidden' : 'visible';

      const nextBtn = document.getElementById('next-btn');

      nextBtn.disabled = !answers[q.id];

      document.getElementById('next-btn-text').textContent = currentStep === questions.length - 1 ? 'Ver Resultado' : 'Próxima';

    }

    let qrCodeInstance = null;

    function getShareableUrl() {

      const currentUrl = window.location.href;

      // Verifica se está em ambiente local/sandbox (about:srcdoc, blob:, file:)

      if (!currentUrl || currentUrl.startsWith('about:') || currentUrl.startsWith('blob:') || currentUrl.startsWith('file:')) {

        return {

          url: '',

          isLocal: true

        };

      }

      return { url: currentUrl, isLocal: false };

    }

    function renderQRCode(targetUrl) {

      const qrBox = document.getElementById('qrcode-box');

      qrBox.innerHTML = '';

      const validText = targetUrl && targetUrl.trim() !== '' ? targetUrl.trim() : 'https://';

      qrCodeInstance = new QRCode(qrBox, {

        text: validText,

        width: 180,

        height: 180,

        colorDark: '#000000',

        colorLight: '#ffffff',

        correctLevel: QRCode.CorrectLevel.H

      });

    }

    function updateQRCodeFromInput() {

      const urlInput = document.getElementById('share-url-input');

      renderQRCode(urlInput.value);

    }

    function openShareModal() {

      const modal = document.getElementById('share-modal');

      const urlInput = document.getElementById('share-url-input');

      const localNotice = document.getElementById('local-preview-notice');

      const copyFeedback = document.getElementById('copy-feedback');

      copyFeedback.classList.add('hidden');

      const { url, isLocal } = getShareableUrl();

     

      if (!urlInput.value) {

        urlInput.value = url;

      }

      if (isLocal) {

        localNotice.classList.remove('hidden');

      } else {

        localNotice.classList.add('hidden');

      }

      renderQRCode(urlInput.value);

      modal.classList.remove('hidden');

    }

    function closeShareModal() {

      document.getElementById('share-modal').classList.add('hidden');

    }

    function copyShareLink() {

      const urlInput = document.getElementById('share-url-input');

      const copyFeedback = document.getElementById('copy-feedback');

      const copyBtn = document.getElementById('copy-link-btn');

      urlInput.select();

      urlInput.setSelectionRange(0, 99999);

      document.execCommand('copy');

      copyFeedback.classList.remove('hidden');

      copyBtn.textContent = 'Copiado!';

      copyBtn.classList.replace('bg-red-600', 'bg-green-600');

      setTimeout(() => {

        copyBtn.textContent = 'Copiar';

        copyBtn.classList.replace('bg-green-600', 'bg-red-600');

      }, 2500);

    }

    function downloadQRCode() {

      const qrCanvas = document.querySelector('#qrcode-box canvas');

      const qrImg = document.querySelector('#qrcode-box img');

      const link = document.createElement('a');

      link.download = 'qrcode-simulador-cotas.png';

      if (qrCanvas) {

        link.href = qrCanvas.toDataURL('image/png');

        link.click();

      } else if (qrImg && qrImg.src) {

        link.href = qrImg.src;

        link.click();

      }

    }
